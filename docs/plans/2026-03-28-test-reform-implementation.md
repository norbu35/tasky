# Test Reform Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Implement the three-layer test reform: scenario files (human-authored specs) → test code (agent-implemented) → registry + CI gates (automated enforcement).

**Architecture:** Scenario files in `tests/scenarios/` are the source of truth for what must be tested. A sync script generates `tests/registry.yaml` from those files and cross-references test implementations. Three CI gates (Smoke, Regression, Full Quality) enforce coverage by risk tier automatically.

**Tech Stack:** Java 21, JUnit 5, AssertJ, Gradle, Bash (scripts), YAML (registry), existing PIT + JaCoCo setup.

**Design doc:** `docs/plans/2026-03-28-test-reform-design.md` — read this before starting.

### Task 2: Write `scripts/sync-registry.sh`

**Files:**
- Create: `scripts/sync-registry.sh`
- Create: `tests/registry.yaml` (empty seed)

**Step 1: Write the sync script**

```bash
#!/usr/bin/env bash
# sync-registry.sh
# Parses tests/scenarios/*.md for SCN-* entries, syncs tests/registry.yaml,
# then cross-references @DisplayName annotations in test code to update status,
# then parses PIT XML to update mutation_kill_rate per domain.
set -euo pipefail

SCENARIOS_DIR="tests/scenarios"
REGISTRY="tests/registry.yaml"
TEST_SRC="src/test/java"
PIT_XML="build/reports/pitest/mutations.xml"

# ── Step 1: Parse scenario files ─────────────────────────────────────────────
python3 - <<'PYEOF'
import re, yaml, os, sys
from pathlib import Path

scenarios_dir = Path("tests/scenarios")
registry_path = Path("tests/registry.yaml")

# Load existing registry to preserve notes and override_status
existing = {}
if registry_path.exists():
    with open(registry_path) as f:
        data = yaml.safe_load(f) or {}
        existing = data.get("scenarios", {})

scenarios = {}

for md_file in sorted(scenarios_dir.glob("*.md")):
    domain = md_file.stem
    if domain in ("README",):
        continue
    content = md_file.read_text()
    # Match ## SCN-XXX-NNN blocks
    blocks = re.split(r'\n(?=## SCN-)', content)
    for block in blocks:
        m = re.match(r'## (SCN-[A-Z]+-\d+)\n\*\*Risk:\*\* (\w+)\n\*\*PRD:\*\* ([^\n]+)\n\*\*Title:\*\* ([^\n]+)', block)
        if not m:
            continue
        scn_id, risk, prd_ref, title = m.group(1), m.group(2), m.group(3).strip(), m.group(4).strip()

        # Determine test_type from risk
        test_type = "domain-unit" if risk in ("critical", "high") else "medium-unit"
        if domain == "integration":
            test_type = "integration"

        prev = existing.get(scn_id, {})
        scenarios[scn_id] = {
            "title": title,
            "domain": domain,
            "risk": risk.lower(),
            "prd_ref": prd_ref,
            "test_type": test_type,
            "status": prev.get("status", "untested"),
            "mutation_kill_rate": prev.get("mutation_kill_rate", None),
            "notes": prev.get("notes", None),
            "override_status": prev.get("override_status", None),
        }

with open(registry_path, "w") as f:
    f.write("# Auto-generated by scripts/sync-registry.sh\n")
    f.write("# Edit scenario entries in tests/scenarios/<domain>.md only.\n")
    f.write("# Hand-edit only: notes, override_status.\n\n")
    yaml.dump({"scenarios": scenarios}, f, default_flow_style=False, sort_keys=True, allow_unicode=True)

print(f"[sync] Parsed {len(scenarios)} scenarios from {scenarios_dir}")
PYEOF

# ── Step 2: Scan test @DisplayName annotations for SCN-* IDs ─────────────────
python3 - <<'PYEOF'
import re, yaml
from pathlib import Path

registry_path = Path("tests/registry.yaml")
test_src = Path("src/test/java")

with open(registry_path) as f:
    data = yaml.safe_load(f)
scenarios = data["scenarios"]

# Find all SCN-* references in @DisplayName annotations
found_ids = set()
pattern = re.compile(r'@DisplayName\s*\(\s*["\']+(SCN-[A-Z]+-\d+)[^"\']*["\']')
for java_file in test_src.rglob("*.java"):
    content = java_file.read_text(errors="ignore")
    for m in pattern.finditer(content):
        found_ids.add(m.group(1))

# Update status
updated = 0
for scn_id, entry in scenarios.items():
    if entry.get("override_status"):
        continue
    new_status = "covered" if scn_id in found_ids else "untested"
    if entry["status"] != new_status:
        entry["status"] = new_status
        updated += 1

with open(registry_path, "w") as f:
    f.write("# Auto-generated by scripts/sync-registry.sh\n")
    f.write("# Edit scenario entries in tests/scenarios/<domain>.md only.\n")
    f.write("# Hand-edit only: notes, override_status.\n\n")
    yaml.dump({"scenarios": scenarios}, f, default_flow_style=False, sort_keys=True, allow_unicode=True)

print(f"[sync] Updated status for {updated} scenarios. Covered: {sum(1 for e in scenarios.values() if e['status'] == 'covered')}/{len(scenarios)}")
PYEOF

# ── Step 3: Parse PIT XML for mutation_kill_rate per domain ──────────────────
if [ -f "$PIT_XML" ]; then
python3 - <<'PYEOF'
import xml.etree.ElementTree as ET, yaml
from collections import defaultdict
from pathlib import Path

registry_path = Path("tests/registry.yaml")
pit_xml = Path("build/reports/pitest/mutations.xml")

with open(registry_path) as f:
    data = yaml.safe_load(f)
scenarios = data["scenarios"]

tree = ET.parse(pit_xml)
root = tree.getroot()

# Map package → (killed, total)
pkg_stats = defaultdict(lambda: {"killed": 0, "total": 0})
for m in root.findall("mutation"):
    cls = m.findtext("mutatedClass", "")
    pkg = ".".join(cls.split(".")[:-1])
    # Map to domain name (mn.tasky.booking.* → booking)
    parts = cls.split(".")
    if len(parts) >= 3:
        domain = parts[2]  # mn.tasky.<domain>
        status = m.get("status", "")
        pkg_stats[domain]["total"] += 1
        if status == "KILLED":
            pkg_stats[domain]["killed"] += 1

# Build domain kill rates
domain_rates = {}
for domain, stats in pkg_stats.items():
    if stats["total"] > 0:
        domain_rates[domain] = round(100 * stats["killed"] / stats["total"])

# Update mutation_kill_rate in scenarios
for scn_id, entry in scenarios.items():
    domain = entry.get("domain", "")
    if domain in domain_rates:
        entry["mutation_kill_rate"] = domain_rates[domain]

with open(registry_path, "w") as f:
    f.write("# Auto-generated by scripts/sync-registry.sh\n")
    f.write("# Edit scenario entries in tests/scenarios/<domain>.md only.\n")
    f.write("# Hand-edit only: notes, override_status.\n\n")
    yaml.dump({"scenarios": scenarios}, f, default_flow_style=False, sort_keys=True, allow_unicode=True)

print(f"[sync] Updated mutation_kill_rate for domains: {list(domain_rates.keys())}")
PYEOF
else
  echo "[sync] PIT XML not found at $PIT_XML — skipping mutation_kill_rate update"
fi

echo "[sync] Done. Registry: tests/registry.yaml"
```

### Task 3: Write `scripts/check-gates.sh`

**Files:**
- Create: `scripts/check-gates.sh`

**Step 1: Write the gate checker**

```bash
#!/usr/bin/env bash
# check-gates.sh
# Reads tests/registry.yaml and enforces quality gates.
# Usage: ./scripts/check-gates.sh <gate>
#   gate: smoke | regression | full
# Exit code: 0 = pass, 1 = fail
set -euo pipefail

GATE="${1:-smoke}"
REGISTRY="tests/registry.yaml"

python3 - "$GATE" "$REGISTRY" <<'PYEOF'
import sys, yaml

gate = sys.argv[1]
registry_path = sys.argv[2]

with open(registry_path) as f:
    data = yaml.safe_load(f)
scenarios = data.get("scenarios", {})

failures = []

if gate in ("smoke", "regression", "full"):
    # Gate 1 (Smoke): all Critical scenarios must be covered
    for scn_id, entry in scenarios.items():
        if entry["risk"] == "critical":
            override = entry.get("override_status") or ""
            if entry["status"] != "covered" and not override.startswith("waived"):
                failures.append(f"[SMOKE] Critical scenario not covered: {scn_id} — {entry['title']}")

if gate in ("regression", "full"):
    # Gate 2 (Regression): all High scenarios must be covered
    for scn_id, entry in scenarios.items():
        if entry["risk"] == "high":
            override = entry.get("override_status") or ""
            if entry["status"] != "covered" and not override.startswith("waived"):
                failures.append(f"[REGRESSION] High scenario not covered: {scn_id} — {entry['title']}")

    # Mutation kill rate floors for Critical and High domains
    FLOORS = {"critical": 75, "high": 60}
    domain_risks = {}
    for entry in scenarios.values():
        d = entry["domain"]
        r = entry["risk"]
        # Worst (lowest) tier wins for the domain
        current = domain_risks.get(d, "low")
        tier_order = {"critical": 0, "high": 1, "medium": 2, "low": 3}
        if tier_order.get(r, 3) < tier_order.get(current, 3):
            domain_risks[d] = r

    for entry in scenarios.values():
        d = entry["domain"]
        domain_risk = domain_risks.get(d, "low")
        floor = FLOORS.get(domain_risk)
        if floor is None:
            continue
        kill_rate = entry.get("mutation_kill_rate")
        if kill_rate is not None and kill_rate < floor:
            failures.append(
                f"[REGRESSION] Domain '{d}' mutation kill rate {kill_rate}% < floor {floor}% (tier: {domain_risk})"
            )

if gate == "full":
    # Gate 3: no Critical/High scenario untested without notes
    for scn_id, entry in scenarios.items():
        if entry["risk"] in ("critical", "high"):
            if entry["status"] == "untested" and not entry.get("notes") and not entry.get("override_status"):
                failures.append(f"[FULL] Untested scenario with no notes: {scn_id} — {entry['title']}")

    # Medium floor
    for entry in scenarios.values():
        d = entry["domain"]
        kill_rate = entry.get("mutation_kill_rate")
        if kill_rate is not None and entry["risk"] == "medium" and kill_rate < 40:
            failures.append(f"[FULL] Domain '{d}' medium mutation kill rate {kill_rate}% < 40%")

if failures:
    print(f"\n❌ Gate '{gate}' FAILED ({len(failures)} issue(s)):\n")
    for f in failures:
        print(f"  • {f}")
    print()
    sys.exit(1)
else:
    print(f"✅ Gate '{gate}' passed.")
    sys.exit(0)
PYEOF
```

## Backend Testing Rules

### Before writing any test
1. Check `tests/registry.yaml` for an existing scenario

### MUST NOT
- Never modify `tests/scenarios/` files
- Never use `@DirtiesContext` — use `@Transactional` rollback
- Never assert only on mock invocation — assert on observable output state
- Never put more than one scenario in one test method
- Controller tests tagged `@Tag("http-mapping")` are the only tests exempt from SCN-* IDs

### When assigned a survived PIT mutation
1. Find it in `build/reports/pitest/index.html`
2. Find the scenario that covers that behavior in `tests/scenarios/`
3. If scenario exists: fix the assertion — it's not precise enough
4. If no scenario: report the gap — do not add a test without a scenario
